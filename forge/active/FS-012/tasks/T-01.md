## Change Requirements

Add the following types:

```typescript
interface TestCase {
  readonly id: string;
  readonly name: string;
  readonly sourceData: string; // raw JSON string
  readonly expectedOutput?: string; // raw JSON string
  readonly createdAt: string; // ISO date
}

type DiffChangeType = 'added' | 'removed' | 'changed';

interface DiffEntry {
  readonly path: string; // dot-notation path
  readonly type: DiffChangeType;
  readonly actual?: unknown;
  readonly expected?: unknown;
}

interface DiffResult {
  readonly entries: readonly DiffEntry[];
  readonly isEqual: boolean;
}

interface PreviewContextValue {
  readonly sourceData: unknown | null;
  readonly isExecuting: boolean;
  readonly lastResult: ExecutionResult | null;
}

type PreviewExecutionState =
  | { status: 'idle' }
  | { status: 'executing' }
  | { status: 'success'; result: ExecutionResult }
  | { status: 'error'; error: string }
  | { status: 'timeout' };
```

Types must be readonly and use strict TypeScript conventions consistent with existing `domain.ts` patterns.
