```tsx
// State
const [isHistoryOpen, setIsHistoryOpen] = useState(false);

// Hooks
const editor = useMappingEditor(mappingId);
const history = useVersionHistory(mappingId, editor.config);

// Restore handler
const handleRestore = (version: number) => {
  const restoreConfig = history.getRestoreConfig(version);
  if (restoreConfig) {
    editor.actions.restore(restoreConfig);
    setIsHistoryOpen(false);
    // Refresh history after save completes (slight delay or effect-based)
  }
};

// In JSX:
<MappingEditorPage
  ...
  onHistoryToggle={() => setIsHistoryOpen(!isHistoryOpen)}
/>
<VersionHistoryDrawer
  isOpen={isHistoryOpen}
  onClose={() => setIsHistoryOpen(false)}
  versions={history.versions}
  isLoading={history.isLoading}
  isEmpty={history.isEmpty}
  selectedVersion={history.selectedVersion}
  onSelectVersion={history.selectVersion}
  currentVersion={editor.version}
>
  {history.selectedDiff && (
    <VersionDiffView
      diff={history.selectedDiff}
      selectedVersion={history.selectedVersion!}
      currentVersion={editor.version}
      hasUnsavedChanges={editor.hasUnsavedChanges}
      onRestore={handleRestore}
      onBack={() => history.selectVersion(null)}
    />
  )}
</VersionHistoryDrawer>
```

## Notes

The `restore` action on `useMappingEditor` is conceptually: "replace the entire working state with this config and immediately save." Implementation:

```typescript
const restore = useCallback(async (restoreConfig: MappingConfig) => {
  // Replace local state
  setRules(restoreConfig.rules);
  // If config options are tracked locally (FS-017), replace those too
  
  // Build the new config with incremented version
  const newVersion = version + 1;
  const fullConfig: MappingConfig = {
    ...restoreConfig,
    id: config?.id,
    projectId: config?.projectId,
    version: newVersion,
  };
  
  // Save immediately
  await adapter.updateMapping(mappingId, fullConfig);
  // ... update states, save version entry, etc.
}, [...]);
```

This approach ensures the restored config keeps the same mapping ID/project context but replaces content.

For the history refresh after restore: use `useEffect` triggered by `version` change, or have the restore action call `history.refresh()` directly after save completes.
