### Source context formatting

Create a utility function (can be inline or in a small helper):

```typescript
function formatSourceContext(parsedSourceSchema: ParsedSchema | null): string {
  if (!parsedSourceSchema?.nodes) return '';
  const lines: string[] = [];
  function walk(nodes: SchemaTreeNode[]) {
    for (const node of nodes) {
      if (node.children && node.children.length > 0) {
        walk(node.children);
      } else {
        // Leaf node
        lines.push(`- ${node.path} (${node.type})`);
        if (lines.length >= 200) return; // cap at 200
      }
    }
  }
  walk(parsedSourceSchema.nodes);
  return lines.join('\n');
}
```

### Generate handler
